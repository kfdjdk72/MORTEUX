CombatTab:AddButton("Remove Punch Anim", function()
    local blacklistedAnimations = { ["rbxassetid://3638729053"] = true, ["rbxassetid://3638767427"] = true }

    local function stopCurrentAnimations()
        local character = LocalPlayer.Character
        if not character or not character:FindFirstChild("Humanoid") then
            return
        else
            local humanoid = character:FindFirstChild("Humanoid")
            for _, track in pairs(humanoid:GetPlayingAnimationTracks()) do
                if track.Animation then
                    local name = track.Name:lower()
                    if blacklistedAnimations[track.Animation.AnimationId] or name:match("punch") or name:match("attack") or name:match("right") then
                        track:Stop()
                    end
                end
            end

            if not _G.AnimBlockConnection then
                _G.AnimBlockConnection = humanoid.AnimationPlayed:Connect(function(track)
                    if track.Animation then
                        local name = track.Name:lower()
                        if blacklistedAnimations[track.Animation.AnimationId] or name:match("punch") or name:match("attack") or name:match("right") then
                            track:Stop()
                        end
                    end
                end)
            end
        end
    end

    stopCurrentAnimations()

    local function setupToolListeners()
        local function monitorTool(tool)
            if tool and (tool.Name == "Punch" or tool.Name:match("Attack") or tool.Name:match("Right")) then
                if not tool:GetAttribute("ActivatedOverride") then
                    tool:SetAttribute("ActivatedOverride", true)
                    local connection = tool.Activated:Connect(function()
                        task.wait(0.05)
                        local character = LocalPlayer.Character
                        if character and character:FindFirstChild("Humanoid") then
                            for _, track in pairs(character.Humanoid:GetPlayingAnimationTracks()) do
                                if track.Animation then
                                    local name = track.Name:lower()
                                    if blacklistedAnimations[track.Animation.AnimationId] or name:match("punch") or name:match("attack") or name:match("right") then
                                        track:Stop()
                                    end
                                end
                            end
                        end
                    end)
                    if not _G.ToolConnections then _G.ToolConnections = {} end
                    _G.ToolConnections[tool] = connection
                end
            end
        end

        for _, item in pairs(LocalPlayer.Backpack:GetChildren()) do
            monitorTool(item)
        end

        local character = LocalPlayer.Character
        if character then
            for _, item in pairs(character:GetChildren()) do
                if item:IsA("Tool") then
                    monitorTool(item)
                end
            end
        end

        if not _G.BackpackAddedConnection then
            _G.BackpackAddedConnection = LocalPlayer.Backpack.ChildAdded:Connect(function(child)
                if child:IsA("Tool") then
                    task.wait(0.1)
                    monitorTool(child)
                end
            end)
        end

        if not _G.CharacterToolAddedConnection and character then
            _G.CharacterToolAddedConnection = character.ChildAdded:Connect(function(child)
                if child:IsA("Tool") then
                    task.wait(0.1)
                    monitorTool(child)
                end
            end)
        end
    end

    setupToolListeners()

    if not _G.AnimMonitorConnection then
        _G.AnimMonitorConnection = RunService.Heartbeat:Connect(function()
            if tick() % 0.5 < 0.01 then
                local character = LocalPlayer.Character
                if character and character:FindFirstChild("Humanoid") then
                    for _, track in pairs(character.Humanoid:GetPlayingAnimationTracks()) do
                        if track.Animation then
                            local name = track.Name:lower()
                            if blacklistedAnimations[track.Animation.AnimationId] or name:match("punch") or name:match("attack") or name:match("right") then
                                track:Stop()
                            end
                        end
                    end
                end
            end
        end)
    end

    if not _G.CharacterAddedConnection then
        _G.CharacterAddedConnection = LocalPlayer.CharacterAdded:Connect(function(newCharacter)
            task.wait(1)
            stopCurrentAnimations()
            setupToolListeners()
            if _G.CharacterToolAddedConnection then
                _G.CharacterToolAddedConnection:Disconnect()
            end
            _G.CharacterToolAddedConnection = newCharacter.ChildAdded:Connect(function(child)
                if child:IsA("Tool") then
                    task.wait(0.1)
                    monitorTool(child)
                end
            end)
        end)
    end
end)

function RecoveryPunch()
    if _G.AnimBlockConnection then
        _G.AnimBlockConnection:Disconnect()
        _G.AnimBlockConnection = nil
    end
    if _G.AnimMonitorConnection then
        _G.AnimMonitorConnection:Disconnect()
        _G.AnimMonitorConnection = nil
    end
    if _G.ToolConnections then
        for tool, connection in pairs(_G.ToolConnections) do
            if connection then connection:Disconnect() end
        end
        _G.ToolConnections = nil
    end
    if _G.BackpackAddedConnection then
        _G.BackpackAddedConnection:Disconnect()
        _G.BackpackAddedConnection = nil
    end
    if _G.CharacterToolAddedConnection then
        _G.CharacterToolAddedConnection:Disconnect()
        _G.CharacterToolAddedConnection = nil
    end
    if _G.CharacterAddedConnection then
        _G.CharacterAddedConnection:Disconnect()
        _G.CharacterAddedConnection = nil
    end
end

local GodPunchActive = false
local GodPunchConnections = {}

CombatTab:AddSwitch("Punch When Dead (Protein Egg Required)", function(state)
    GodPunchActive = state
    for _, conn in pairs(GodPunchConnections) do if conn.Disconnect then conn:Disconnect() end end
    GodPunchConnections = {}

    if state then
        _G.AutoPunchToggle = true
        task.spawn(function()
            local backpack = LocalPlayer:WaitForChild("Backpack")
            while _G.AutoPunchToggle do
                local remote = LocalPlayer:FindFirstChild("muscleEvent")
                local character = LocalPlayer.Character
                if character and character:FindFirstChild("Humanoid") and remote then
                    local punch = backpack:FindFirstChild("Punch")
                    if not character:FindFirstChild("Punch") and punch then
                        character.Humanoid:EquipTool(punch)
                    end
                    remote:FireServer("punch", "rightHand")
                end
                task.wait(0.0001)
            end
        end)

        _G.AutoProteinEgg = true
        local function restoreTransparency(model)
            for _, desc in ipairs(model:GetDescendants()) do
                if desc:IsA("BasePart") then
                    desc.Transparency = 0
                    pcall(function() desc.LocalTransparencyModifier = 0 end)
                end
            end
        end

        local function getEgg()
            return LocalPlayer.Backpack:FindFirstChild("Protein Egg") or StarterPack:FindFirstChild("Protein Egg") or ReplicatedStorage:FindFirstChild("Protein Egg")
        end

        local function useEgg(egg)
            local character = LocalPlayer.Character
            if not character or not character:FindFirstChild("Humanoid") then return end
            pcall(function() character.Humanoid:EquipTool(egg) end)
            task.wait(0.1)
            if not character:FindFirstChild("Protein Egg") then egg.Parent = character task.wait(0.1) end
            local equippedEgg = character:FindFirstChild("Protein Egg")
            if equippedEgg then restoreTransparency(equippedEgg) end
        end

        local function handleEggCycle()
            local character = LocalPlayer.Character
            if not _G.AutoProteinEgg or not character then return end
            local egg = character:FindFirstChild("Protein Egg")
            local needsNewEgg = not egg
            if egg then
                for _, part in ipairs(egg:GetDescendants()) do
                    if part:IsA("BasePart") and part.Transparency > 0 then needsNewEgg = true break end
                end
            end
            if needsNewEgg then
                local targetEgg = getEgg()
                if targetEgg then
                    if targetEgg.Parent ~= LocalPlayer.Backpack then
                        local clone = targetEgg:Clone()
                        clone.Parent = LocalPlayer.Backpack
                        targetEgg = clone
                    end
                    useEgg(targetEgg)
                end
            end
        end

        LocalPlayer.CharacterAdded:Connect(function(char) task.wait(1) handleEggCycle() end)
        LocalPlayer.Backpack.ChildAdded:Connect(function(child) if _G.AutoProteinEgg and child.Name == "Protein Egg" then task.wait(0.2) handleEggCycle() end end)
        task.spawn(function() while _G.AutoProteinEgg do handleEggCycle() task.wait(0.5) end end)

        getgenv().AntiFlyActive = true
        GodPunchConnections.AntiFly = RunService.Heartbeat:Connect(function()
            local character = LocalPlayer.Character
            if not character then return end
            local root = character:FindFirstChild("HumanoidRootPart")
            local hum = character:FindFirstChildOfClass("Humanoid")
            if root and hum then
                local ray = Ray.new(root.Position, Vector3.new(0, -500, 0))
                local hit, pos = Workspace:FindPartOnRay(ray, character)
                if hit then
                    local y = pos.Y
                    if root.Position.Y - y > 0.5 then
                        root.CFrame = CFrame.new(root.Position.X, y + 0.5, root.Position.Z)
                        hum.PlatformStand = true
                        hum.PlatformStand = false
                    end
                end
            end
        end)

        local function clearParticles()
            local effects = { ParticleEmitter = true, Trail = true, Smoke = true, Fire = true }
            for _, obj in ipairs(Workspace:GetChildren()) do
                if obj:IsA("Model") or obj:IsA("Part") then
                    for _, child in ipairs(obj:GetChildren()) do
                        if effects[child.ClassName] then pcall(function() child:Destroy() end) end
                    end
                end
            end
            local terrain = Workspace:FindFirstChildOfClass("Terrain")
            if terrain then
                terrain.WaterWaveSize = 0
                terrain.WaterReflectance = 0
                terrain.WaterTransparency = 1
            end
        end

        local function optimizeLighting()
            Lighting.ClockTime = 18
            Lighting.Brightness = 1.5
            Lighting.OutdoorAmbient = Color3.fromRGB(150, 100, 80)
            Lighting.FogColor = Color3.fromRGB(200, 120, 100)
            Lighting.FogEnd = 500
            for _, obj in ipairs(Lighting:GetChildren()) do if obj:IsA("Sky") then obj:Destroy() end end
            local sky = Instance.new("Sky")
            sky.Name = "SunsetSky"
            sky.SkyboxBk = "rbxassetid://131889017"
            sky.SkyboxDn = "rbxassetid://131889017"
            sky.SkyboxFt = "rbxassetid://131889017"
            sky.SkyboxLf = "rbxassetid://131889017"
            sky.SkyboxRt = "rbxassetid://131889017"
            sky.SkyboxUp = "rbxassetid://131889017"
            sky.Parent = Lighting
        end

        clearParticles()
        optimizeLighting()

        task.spawn(function()
            while GodPunchActive do
                local shake = LocalPlayer.Backpack:FindFirstChild("Tropical Shake")
                if not shake then break end
                shake.Parent = LocalPlayer.Character
                RunService.Heartbeat:Wait()
                if shake.Activate then shake:Activate() elseif mouse1click then mouse1click() end
                task.wait(0.1)
            end
        end)
    else
        _G.AutoPunchToggle = false
        _G.AutoProteinEgg = false
        getgenv().AntiFlyActive = false
        if GodPunchConnections.AntiFly then GodPunchConnections.AntiFly:Disconnect() GodPunchConnections.AntiFly = nil end
    end
end)

local function SetupRockFarm(toggle, durabilityRequirement)
    getgenv().autoFarm = toggle
    task.spawn(function()
        while getgenv().autoFarm do
            task.wait()
            if not getgenv().autoFarm then break end
            if LocalPlayer.Durability.Value >= durabilityRequirement then
                for _, machine in pairs(Workspace.machinesFolder:GetDescendants()) do
                    if machine.Name == "neededDurability" and machine.Value == durabilityRequirement then
                        local rock = machine.Parent:FindFirstChild("Rock")
                        local char = LocalPlayer.Character
                        if rock and char and char:FindFirstChild("LeftHand") and char:FindFirstChild("RightHand") then
                            firetouchinterest(rock, char.RightHand, 0)
                            firetouchinterest(rock, char.RightHand, 1)
                            firetouchinterest(rock, char.LeftHand, 0)
                            firetouchinterest(rock, char.LeftHand, 1)
                            AutoPunchTool()
                        end
                    end
                end
            end
        end
    end)
end

        if obj:IsA("Decal") or obj:IsA("Texture") then obj.Transparency = 1
        elseif obj:IsA("BasePart") and not obj:IsA("MeshPart") then
            obj.Material = Enum.Material.SmoothPlastic
            obj.Reflectance = 0
        elseif obj:IsA("ParticleEmitter") or obj:IsA("Fire") or obj:IsA("Smoke") or obj:IsA("Sparkles") then obj.Enabled = false
        end
    end
    Lighting.GlobalShadows = false
    Lighting.FogEnd = 9e9
    settings().Rendering.QualityLevel = 1
end)

local MiscTab = Window:AddTab("Miscs")
local Misc1 = MiscTab:AddFolder("Miscellaneous 1")

Misc1:AddSwitch("Auto Punch", function(state)
    _G.FastPunch = state
    if state then
        task.spawn(function()
            while _G.FastPunch do
                local char = LocalPlayer.Character
                if char then
                    local punch = char:FindFirstChild("Punch") or LocalPlayer.Backpack:FindFirstChild("Punch")
                    if punch then
                        if not char:FindFirstChild("Punch") then char.Humanoid:EquipTool(punch) end
                        if punch:FindFirstChild("attackTime") then punch.attackTime.Value = 0 end
                    end
                end
                task.wait(0.1)
            end
        end)
        task.spawn(function()
            while _G.FastPunch do
                LocalPlayer.muscleEvent:FireServer("punch", "rightHand")
                LocalPlayer.muscleEvent:FireServer("punch", "leftHand")
                if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Punch") then LocalPlayer.Character.Punch:Activate() end
                task.wait(0)
            end
        end)
    end
end)

local StatsTab = Window:AddTab("Stats")
local SelectedStatPlayer = ""
local PlayerSelectDropdown = StatsTab:AddDropdown("Select Player", function(name) SelectedStatPlayer = name:match("| (.+)") or name end)
for _, p in ipairs(Players:GetPlayers()) do PlayerSelectDropdown:Add(p.DisplayName .. " | " .. p.Name) end

local StrengthStat = StatsTab:AddLabel("Strength: -")
local AgilityStat = StatsTab:AddLabel("Agility: -")
local DurabilityStat = StatsTab:AddLabel("Durability: -")

task.spawn(function()
    while true do
        local target = Players:FindFirstChild(SelectedStatPlayer)
        if target then
            local lstats = target:FindFirstChild("leaderstats")
            StrengthStat.Text = "Strength: " .. (lstats and lstats:FindFirstChild("Strength") and lstats.Strength.Value or 0)
            AgilityStat.Text = "Agility: " .. (target:FindFirstChild("Agility") and target.Agility.Value or 0)
            DurabilityStat.Text = "Durability: " .. (target:FindFirstChild("Durability") and target.Durability.Value or 0)
        end
        task.wait(1)
    end
end)
