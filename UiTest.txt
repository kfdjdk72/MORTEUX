local function Create(className, properties, parent)
	local object = Instance.new(className)

	for property, value in pairs(properties or {}) do
		object[property] = value
	end

	object.Parent = parent

	return object
end

local function AddCorner(object, radius)
	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(0, radius)
	corner.Parent = object
	return corner
end

local function AddStroke(object, thickness, color)
	local stroke = Instance.new("UIStroke")
	stroke.Thickness = thickness or 2
	stroke.Color = color or UI.Theme.Black
	stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
	stroke.Parent = object
	return stroke
end

local function AddGradient(object, firstColor, secondColor, rotation)
	local gradient = Instance.new("UIGradient")
	gradient.Color = ColorSequence.new(firstColor, secondColor)
	gradient.Rotation = rotation or 90
	gradient.Parent = object
	return gradient
end

local function StyleText(object, font, size, color, strokeTransparency)
	object.Font = font
	object.TextSize = size
	object.TextColor3 = color or UI.Theme.White
	object.TextStrokeColor3 = UI.Theme.Black
	object.TextStrokeTransparency = strokeTransparency == nil and 0 or strokeTransparency
end

local function Animate(object, duration, properties, style, direction)
	local tween = TweenService:Create(
		object,
		TweenInfo.new(
			duration or 0.15,
			style or Enum.EasingStyle.Quad,
			direction or Enum.EasingDirection.Out
		),
		properties
	)

	tween:Play()

	return tween
end

local function MakeDraggable(handle, target)
	local dragging = false
	local dragStart
	local startPosition

	handle.InputBegan:Connect(function(input)
		if input.UserInputType ~= Enum.UserInputType.MouseButton1
			and input.UserInputType ~= Enum.UserInputType.Touch then
			return
		end

		dragging = true
		dragStart = input.Position
		startPosition = target.Position

		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
			end
		end)
	end)

	UserInputService.InputChanged:Connect(function(input)
		if not dragging then
			return
		end

		if input.UserInputType ~= Enum.UserInputType.MouseMovement
			and input.UserInputType ~= Enum.UserInputType.Touch then
			return
		end

		local delta = input.Position - dragStart

		target.Position = UDim2.new(
			startPosition.X.Scale,
			startPosition.X.Offset + delta.X,
			startPosition.Y.Scale,
			startPosition.Y.Offset + delta.Y
		)
	end)
end

	local function UpdateResponsive()
		local camera = workspace.CurrentCamera

		if not camera then
			return
		end

		local viewport = camera.ViewportSize

		local width = math.min(820, viewport.X - 30)
		local height = math.min(590, viewport.Y - 30)

		if viewport.X < 700 then
			width = math.max(320, viewport.X - 20)
			height = math.max(300, math.min(590, viewport.Y - 20))
		end

		self.Window.Size = UDim2.fromOffset(width, height)

		if width < 520 then
			self.TitlePlate.Size = UDim2.fromOffset(
				math.min(280, width - 100),
				68
			)

			self.Title.TextSize = 25
		else
			self.TitlePlate.Size = UDim2.fromOffset(335, 76)
			self.Title.TextSize = 31
		end
	end

	self.Connections[#self.Connections + 1] =
		workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(
			function()
				if workspace.CurrentCamera then
					UpdateResponsive()
				end
			end
		)

	if workspace.CurrentCamera then
		self.Connections[#self.Connections + 1] =
			workspace.CurrentCamera:GetPropertyChangedSignal("ViewportSize"):Connect(
				UpdateResponsive
			)
	end

	UpdateResponsive()

	return self
end

function UI:SetVisible(value)
	self.Gui.Enabled = value == true
end

function UI:Toggle()
	self:SetVisible(not self.Gui.Enabled)
end

function UI:SetTitle(value)
	self.TitleText = tostring(value)
	self.Title.Text = self.TitleText
end

function UI:SetSize(width, height)
	self.Window.Size = UDim2.fromOffset(width, height)
end

function UI:SetPosition(position)
	self.Window.Position = position
end

function UI:SetPattern(asset)
	self.Pattern.Image = asset
end

function UI:SetPatternTransparency(value)
	self.Pattern.ImageTransparency = math.clamp(value, 0, 1)
end

	function tab:CreateGroup(groupName, settings)
		return self.UI:CreateGroup(tab, groupName, settings)
	end

	tab.UI = self

	return tab

function UI:SelectTab(tab)
	if not tab then
		return
	end

	self.ActiveTab = tab

	for _, current in ipairs(self.Tabs) do
		local active = current == tab

		current.Page.Visible = active

		if active then
			Animate(
				current.Button,
				0.18,
				{
					BackgroundColor3 = current.Color
				}
			)
		else
			Animate(
				current.Button,
				0.18,
				{
					BackgroundColor3 = UI.Theme.Surface2
				}
			)
		end
	end
end

		options = options or {}

		minimum = tonumber(minimum) or 0
		maximum = tonumber(maximum) or 100
		default = tonumber(default) or minimum

		if maximum < minimum then

		AddStroke(row, 3, UI.Theme.Black)

		local titleLabel = Create("TextLabel", {
			Name = "Label",
			Size = UDim2.new(0.55, 0, 0, 31),

			BackgroundTransparency = 1,
			Text = tostring(text),
			TextXAlignment = Enum.TextXAlignment.Left,
			TextYAlignment = Enum.TextYAlignment.Center,
			ZIndex = 16
		}, row)

		StyleText(
			titleLabel,
			UI.Fonts.Title,
			options.TextSize or 17,
			UI.Theme.White,
			0
		)

		local valueLabel = Create("TextLabel", {
			Name = "Value",
			Size = UDim2.fromOffset(90, 31),
			Position = UDim2.new(1, -101, 0, 6),
			BackgroundTransparency = 1,
			Text = tostring(default),
			TextXAlignment = Enum.TextXAlignment.Right,
			TextYAlignment = Enum.TextYAlignment.Center,
			ZIndex = 16
		}, row)

		StyleText(
			valueLabel,
			UI.Fonts.Title,
			options.ValueTextSize or 17,
			UI.Theme.Yellow,
			0
		)

		local track = Create("Frame", {
			Name = "Track",
			Size = UDim2.new(1, -22, 0, 13),
			Position = UDim2.new(0, 11, 1, -25),
			BackgroundColor3 = UI.Theme.Gray,
			BorderSizePixel = 0,
			ZIndex = 16
		}, row)

		AddCorner(track, 7)
		AddStroke(track, 2, UI.Theme.Black)

		if maximum ~= minimum then
			percentage = (default - minimum) / (maximum - minimum)
		end

		local fill = Create("Frame", {
			Name = "Fill",
			Size = UDim2.new(percentage, 0, 1, 0),
			Position = UDim2.fromScale(0, 0),

			BorderSizePixel = 0,
			ZIndex = 17
		}, track)

		AddCorner(fill, 7)

		AddGradient(
			fill,
			Color3.fromRGB(255, 218, 78),
			UI.Theme.YellowDark,
			0
		)

		local handle = Create("Frame", {
			Name = "Handle",
			Size = UDim2.fromOffset(23, 23),
			AnchorPoint = Vector2.new(0.5, 0.5),
			Position = UDim2.new(percentage, 0, 0.5, 0),
			BackgroundColor3 = UI.Theme.Green,
			BorderSizePixel = 0,
			ZIndex = 20
		}, track)

		AddCorner(handle, 6)
		AddStroke(handle, 3, UI.Theme.Black)

		AddGradient(
			handle,
			Color3.fromRGB(100, 255, 105),
			UI.Theme.GreenDark,
			90
		)

		local hitbox = Create("TextButton", {
			Name = "Hitbox",
			Size = UDim2.new(1, 20, 1, 30),
			Position = UDim2.fromOffset(-10, -15),
			BackgroundTransparency = 1,
			BorderSizePixel = 0,
			Text = "",
			AutoButtonColor = false,
			ZIndex = 25
		}, track)

		local value = default
		local dragging = false

		local function SetValue(newValue, invokeCallback, animated)
			newValue = tonumber(newValue) or minimum
			newValue = math.clamp(newValue, minimum, maximum)

			if options.Step then
				local step = tonumber(options.Step) or 1

				if step > 0 then
					newValue =
						math.floor(
							((newValue - minimum) / step) + 0.5
						) * step + minimum

					newValue = math.clamp(
						newValue,
						minimum,
						maximum
					)
				end
			end

			value = newValue

			local alpha = 0

			if maximum ~= minimum then
				alpha = (value - minimum) / (maximum - minimum)
			end

			valueLabel.Text = tostring(value)

			local fillProperties = {
				Size = UDim2.new(alpha, 0, 1, 0)
			}

			local handleProperties = {
				Position = UDim2.new(alpha, 0, 0.5, 0)
			}

			if animated then
				Animate(
					fill,
					0.1,
					fillProperties,
					Enum.EasingStyle.Quad,
					Enum.EasingDirection.Out
				)

				Animate(
					handle,
					0.1,
					handleProperties,
					Enum.EasingStyle.Quad,
					Enum.EasingDirection.Out
				)
			else
				fill.Size = fillProperties.Size
				handle.Position = handleProperties.Position
			end

			if invokeCallback and callback then
				task.spawn(callback, value)
			end
		end

		local function UpdateFromInput(input)
			if track.AbsoluteSize.X <= 0 then
				return
			end

			local x = input.Position.X
			local left = track.AbsolutePosition.X
			local width = track.AbsoluteSize.X

			local alpha = math.clamp(
				(x - left) / width,
				0,
				1
			)

			local newValue =
				minimum + ((maximum - minimum) * alpha)

			SetValue(
				newValue,
				true,
				false
			)
		end

		hitbox.MouseEnter:Connect(function()
			Animate(
				handle,
				0.1,
				{
					Size = UDim2.fromOffset(27, 27)
				}
			)
		end)

		hitbox.MouseLeave:Connect(function()
			if not dragging then
				Animate(
					handle,
					0.1,
					{
						Size = UDim2.fromOffset(23, 23)
					}
				)
			end
		end)

		hitbox.MouseButton1Down:Connect(function()
			dragging = true

			Animate(
				handle,
				0.08,
				{
					Size = UDim2.fromOffset(29, 29)
				}
			)

			local mousePosition = UserInputService:GetMouseLocation()

			UpdateFromInput({
				Position = mousePosition
			})
		end)

		UserInputService.InputChanged:Connect(function(input)
			if not dragging then
				return
			end

			if input.UserInputType == Enum.UserInputType.MouseMovement then
				UpdateFromInput(input)
			elseif input.UserInputType == Enum.UserInputType.Touch then
				UpdateFromInput(input)
			end
		end)

		UserInputService.InputEnded:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1
				or input.UserInputType == Enum.UserInputType.Touch then

				if dragging then
					dragging = false

					Animate(
						handle,
						0.1,
						{
							Size = UDim2.fromOffset(23, 23)
						}
					)
				end
			end
		end)

		SetValue(default, false, false)

		group.Elements[#group.Elements + 1] = row

		return {
			Container = row,
			Label = titleLabel,
			ValueLabel = valueLabel,
			Track = track,
			Fill = fill,
			Handle = handle,

			Set = function(_, newValue)
				SetValue(newValue, false, true)
			end,

			Get = function()
				return value
			end,

			SetMinMax = function(_, newMinimum, newMaximum)
				minimum = tonumber(newMinimum) or minimum
				maximum = tonumber(newMaximum) or maximum

				if maximum < minimum then
					minimum, maximum = maximum, minimum
				end

				SetValue(value, false, false)
			end
		}
	end

			opened = false

			Animate(
				optionFrame,
				0.16,

				}

			Animate(
				arrow,
				0.16,
				{
					Rotation = 0
				}

			task.delay(0.17, function()
				if not opened then
					optionFrame.Visible = false
				end
			end)
		end

		local function OpenDropdown()
			if opened then
				CloseDropdown()
				return
			end

			opened = true
			optionFrame.Visible = true

			local contentHeight =
				optionLayout.AbsoluteContentSize.Y + 10

			Animate(
				optionFrame,
				0.18,
				{
					Size = UDim2.new(
						1,
						0,
						0,
						math.min(contentHeight, options.MaxHeight or 180)

				arrow,
				0.18,

					Rotation = 180
				}

					0.1,
					{
						BackgroundColor3 = UI.Theme.Surface2
					}
				)
			end)

			optionButton.MouseButton1Click:Connect(function()
				selected = optionValue
				selector.Text = tostring(selected)

				CloseDropdown()

				end
			end)
		end

					optionLayout.AbsoluteContentSize.Y + 10

					optionFrame,
					0.1,

			Animate(
				selector,
				0.1,
				{
					BackgroundColor3 = UI.Theme.Surface
				}
			)
		end)

		selector.MouseLeave:Connect(function()
			Animate(
				selector,
				0.1,
				{
					BackgroundColor3 = UI.Theme.Black
				}
			)
		end)

		selector.MouseButton1Click:Connect(OpenDropdown)

		group.Elements[#group.Elements + 1] = row

		return {
			Container = row,
			Label = label,
			Selector = selector,

			Set = function(_, value)
				selected = value
				selector.Text = tostring(value)

				if callback then
					task.spawn(callback, selected)
				end
			end,

			Get = function()
				return selected
			end,

			Refresh = function(_, newValues)
				for _, child in ipairs(optionFrame:GetChildren()) do
					if child:IsA("TextButton") then
						child:Destroy()
					end
				end

				for index, optionValue in ipairs(newValues or {}) do
					local optionButton = Create("TextButton", {
						Name = "Option_" .. tostring(index),
						Size = UDim2.new(1, 0, 0, 34),
						BackgroundColor3 = UI.Theme.Surface2,
						BorderSizePixel = 0,
						Text = tostring(optionValue),
						AutoButtonColor = false,
						LayoutOrder = index,
						ZIndex = 105
					}, optionFrame)

					AddCorner(optionButton, 5)

					StyleText(
						optionButton,
						UI.Fonts.Heading,
						13,
						UI.Theme.White,
						0
					)

					optionButton.MouseButton1Click:Connect(function()
						selected = optionValue
						selector.Text = tostring(selected)

						CloseDropdown()

						if callback then
							task.spawn(callback, selected)
						end
					end)
				end
			end
		}
	end

		BackgroundTransparency = settings.BackgroundTransparency or 0.08,

	}, container)

	for _, frame in ipairs({left, right}) do
		AddCorner(frame, 8)

			local function SetValue(
				newValue,
				invokeCallback,
				animated
			)
				newValue = tonumber(newValue) or minimum

				newValue = math.clamp(
					newValue,
					minimum,
					maximum
				)

				if options.Step then
					local step = tonumber(options.Step) or 1

					if step > 0 then
						newValue =
							math.floor(
								(
									(newValue - minimum)
									/
									step
								) + 0.5
							)
							* step
							+ minimum

						newValue = math.clamp(
							newValue,
							minimum,
							maximum
						)
					end
				end

				value = newValue

				local alpha = 0

				if maximum ~= minimum then
					alpha =
						(value - minimum)
						/
						(maximum - minimum)
				end

				local displayValue = value

				if options.Decimals == 0 then
					displayValue = math.floor(
						value + 0.5
					)
				elseif options.Decimals then
					local multiplier =
						10 ^ options.Decimals

					displayValue =
						math.floor(
							value * multiplier + 0.5
						)
						/
						multiplier
				end

				valueLabel.Text = tostring(displayValue)

				local fillProperties = {
					Size = UDim2.new(
						alpha,
						0,
						1,
						0
					)
				}

				local handleProperties = {
					Position = UDim2.new(
						alpha,
						0,
						0.5,
						0
					)
				}

				if animated then
					Animate(
						fill,
						0.1,
						fillProperties
					)

					Animate(
						handle,
						0.1,
						handleProperties
					)
				else
					fill.Size =
						fillProperties.Size

					handle.Position =
						handleProperties.Position
				end

				if invokeCallback and callback then
					task.spawn(
						callback,
						value
					)
				end
			end

				local width =
					track.AbsoluteSize.X

				if width <= 0 then
					return
				end

				local left =
					track.AbsolutePosition.X

				local alpha =
					math.clamp(
						(positionX - left) / width,
						0,
						1
					)

				local newValue =
					minimum
					+
					(
						(maximum - minimum)
						* alpha
					)

				SetValue(
					newValue,
					true,
					false
				)
			end

				Size = UDim2.new(
					options.WidthScale or 1,
					options.WidthOffset or 0,
					0,
					options.Height or 45
				),
				BackgroundColor3 =
					options.BackgroundColor
					or UI.Theme.Surface2,
				BorderSizePixel = 0,
				LayoutOrder =
					#group.Elements + 1,
				ClipsDescendants = false,
				ZIndex = 30
			}, group.Content)

			AddCorner(row, 6)
			AddStroke(row, 3, UI.Theme.Black)

			local label = Create("TextLabel", {
				Name = "Label",
				Size = UDim2.new(
					1,
					-185,
					1,
					0
				),
				Position =
					UDim2.fromOffset(
						11,
						0
					),
				BackgroundTransparency = 1,
				Text = tostring(text),
				TextXAlignment =
					Enum.TextXAlignment.Left,
				TextYAlignment =
					Enum.TextYAlignment.Center,
				ZIndex = 32
			}, row)

			StyleText(
				label,
				UI.Fonts.Heading,
				options.TextSize or 15,
				UI.Theme.White,
				0
			)

			local selector = Create("TextButton", {
				Name = "Selector",
				Size = UDim2.fromOffset(
					155,
					33
				),
				Position = UDim2.new(
					1,
					-164,
					0.5,
					-16
				),
				BackgroundColor3 =
					UI.Theme.Black,
				BorderSizePixel = 0,
				Text =
					tostring(
						selected or "Select"
					),
				AutoButtonColor = false,
				ZIndex = 33
			}, row)

			AddCorner(selector, 5)
			AddStroke(
				selector,
				2,
				UI.Theme.Black
			)

			StyleText(
				selector,
				UI.Fonts.Heading,
				options.ValueTextSize or 13,
				UI.Theme.Yellow,
				0
			)

			local arrow = Create("TextLabel", {
				Name = "Arrow",
				Size = UDim2.fromOffset(
					25,
					25
				),
				Position = UDim2.new(
					1,
					-28,
					0.5,
					-12
				),
				BackgroundTransparency = 1,
				Text = "▼",
				TextXAlignment =
					Enum.TextXAlignment.Center,
				TextYAlignment =
					Enum.TextYAlignment.Center,
				ZIndex = 34
			}, selector)

			StyleText(
				arrow,
				UI.Fonts.Title,
				11,
				UI.Theme.White,
				0
			)

			local optionFrame = Create("Frame", {
				Name = "Options",
				Size = UDim2.new(
					1,
					0,
					0,
					0
				),
				Position = UDim2.new(
					0,
					0,
					1,
					5
				),
				BackgroundColor3 =
					UI.Theme.Background,
				BorderSizePixel = 0,
				Visible = false,
				ClipsDescendants = true,
				ZIndex = 100
			}, row)

			AddCorner(optionFrame, 6)
			AddStroke(
				optionFrame,
				3,
				UI.Theme.Black
			)

			local optionLayout = Create("UIListLayout", {
				SortOrder = Enum.SortOrder.LayoutOrder,
				Padding = UDim.new(0, 3)
			}, optionFrame)

			Create("UIPadding", {
				PaddingTop = UDim.new(0, 5),
				PaddingBottom = UDim.new(0, 5),
				PaddingLeft = UDim.new(0, 5),
				PaddingRight = UDim.new(0, 5)
			}, optionFrame)

			local opened = false

			local function CloseDropdown()
				opened = false

				Animate(
					optionFrame,
					0.16,
					{
						Size = UDim2.new(
							1,
							0,
							0,
							0
						)
					}
				)

				Animate(
					arrow,
					0.16,
					{
						Rotation = 0
					}
				)

				task.delay(0.17, function()
					if not opened then
						optionFrame.Visible = false
					end
				end)
			end

			local function OpenDropdown()
				if opened then
					CloseDropdown()
					return
				end

				opened = true
				optionFrame.Visible = true

				local contentHeight =
					optionLayout.AbsoluteContentSize.Y + 10

				Animate(
					optionFrame,
					0.18,
					{
						Size = UDim2.new(
							1,
							0,
							0,
							math.min(
								contentHeight,
								options.MaxHeight or 180
							)
						)
					},
					Enum.EasingStyle.Quint,
					Enum.EasingDirection.Out
				)

				Animate(
					arrow,
					0.18,
					{
						Rotation = 180
					}
				)
			end

			local function CreateOption(
				index,
				optionValue
			)
				local optionButton = Create("TextButton", {
					Name =
						"Option_"
						.. tostring(index),
					Size = UDim2.new(
						1,
						0,
						0,
						34
					),
					BackgroundColor3 =
						UI.Theme.Surface2,
					BorderSizePixel = 0,
					Text = tostring(optionValue),
					AutoButtonColor = false,
					LayoutOrder = index,
					ZIndex = 105
				}, optionFrame)

				AddCorner(
					optionButton,
					5
				)

				StyleText(
					optionButton,
					UI.Fonts.Heading,
					13,
					UI.Theme.White,
					0
				)

				optionButton.MouseEnter:Connect(function()
					Animate(
						optionButton,
						0.1,
						{
							BackgroundColor3 =
								UI.Theme.Surface3
						}
					)
				end)

				optionButton.MouseLeave:Connect(function()
					Animate(
						optionButton,
						0.1,
						{
							BackgroundColor3 =
								UI.Theme.Surface2
						}
					)
				end)

				optionButton.MouseButton1Click:Connect(function()
					selected = optionValue
					selector.Text =
						tostring(selected)

					CloseDropdown()

					if callback then
						task.spawn(
							callback,
							selected
						)
					end
				end)

				return optionButton
			end

			for index, optionValue in ipairs(values) do
				CreateOption(
					index,
					optionValue
				)
			end

			optionLayout:GetPropertyChangedSignal(
				"AbsoluteContentSize"
			):Connect(function()
				if opened then
					local contentHeight =
						optionLayout.AbsoluteContentSize.Y
						+ 10

					Animate(
						optionFrame,
						0.1,
						{
							Size = UDim2.new(
								1,
								0,
								0,
								math.min(
									contentHeight,
									options.MaxHeight or 180
								)
							)
						}
					)
				end
			end)

			selector.MouseEnter:Connect(function()
				Animate(
					selector,
					0.1,
					{
						BackgroundColor3 =
							UI.Theme.Surface
					}
				)
			end)

			selector.MouseLeave:Connect(function()
				Animate(
					selector,
					0.1,
					{
						BackgroundColor3 =
							UI.Theme.Black
					}
				)
			end)

			selector.MouseButton1Click:Connect(
				OpenDropdown
			)

			group.Elements[#group.Elements + 1] =
				row

			return {
				Container = row,
				Label = label,
				Selector = selector,
				Options = optionFrame,

				Set = function(_, value)
					selected = value
					selector.Text =
						tostring(value)

					if callback then
						task.spawn(
							callback,
							selected
						)
					end
				end,

				Get = function()
					return selected
				end,

				Refresh = function(_, newValues)
					for _, child in ipairs(
						optionFrame:GetChildren()
					) do
						if child:IsA(
							"TextButton"
						) then
							child:Destroy()
						end
					end

					for index, optionValue in ipairs(
						newValues or {}
					) do
						CreateOption(
							index,
							optionValue
						)
					end
				end
			}
		end

		function group:AddTextButton(
			text,
			callback,
			options
		)
			return group:AddButton(
				text,
				callback,
				options
			)
		end

		function group:AddCheckBox(
			text,
			default,
			callback,
			options
		)
			return group:AddToggle(
				text,
				default,
				callback,
				options
			)
		end

		return group
	end

	AttachMethods(leftGroup)
	AttachMethods(rightGroup)

	tab.Groups[#tab.Groups + 1] = {
		Container = container,
		Left = leftGroup,
		Right = rightGroup
	}

	return {
		Container = container,
		Left = leftGroup,
		Right = rightGroup,

		SetTitle = function(_, side, title)
			if side == "Left" then
				leftHeader.Text = tostring(title)
			elseif side == "Right" then
				rightHeader.Text = tostring(title)
			end
		end
	}
end

return UI
