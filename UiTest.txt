-- Yardımcılar
local function corner(obj, r)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim(0, r)
    c.Parent = obj
end

local function stroke(obj, color, thick, trans)
    local s = Instance.new("UIStroke")
    s.Color = color or colors.elemBorder
    s.Thickness = thick or 1
    s.Transparency = trans or 0
    s.Parent = obj
end

local function tween(obj, info, props)
    return TweenService:Create(obj, info, props)
end

    -- Bir tab ekleme fonksiyonu (buton + içerik)
    function window:AddTab(label, contentCallback)
        -- Buton oluştur
        local btn = Instance.new("TextButton")
        btn.Parent = tabBar
        btn.Size = UDim2(0, 0, 0, 36)
        btn.AutomaticSize = Enum.AutomaticSize.X
        btn.BackgroundColor3 = colors.bgSide
        btn.BackgroundTransparency = 0.8
        btn.Text = label
        btn.Font = Enum.Font.GothamBold
        btn.TextSize = 16
        btn.TextColor3 = colors.textDim
        btn.ZIndex = 6
        corner(btn, 8)
        stroke(btn, colors.elemBorder, 1, 0.3)

        -- Aktif göstergesi (altı çizgi)
        local indicator = Instance.new("Frame")
        indicator.Parent = btn
        indicator.Size = UDim2(0.8, 0, 0, 3)
        indicator.Position = UDim2(0.1, 0, 1, -3)
        indicator.BackgroundColor3 = colors.accent
        indicator.BackgroundTransparency = 1
        indicator.ZIndex = 7

        -- Buton hover efektleri
        btn.MouseEnter:Connect(function()
            if currentTab ~= btn then
                tween(btn, TweenInfo.new(0.15), {BackgroundTransparency = 0.5}):Play()
                btn.TextColor3 = colors.text
            end
        end)
        btn.MouseLeave:Connect(function()
            if currentTab ~= btn then
                tween(btn, TweenInfo.new(0.15), {BackgroundTransparency = 0.8}):Play()
                btn.TextColor3 = colors.textDim
            end
        end)

        -- İçerik oluştur (container)
        local container = Instance.new("Frame")
        container.Parent = contentArea
        container.Size = UDim2(1, 0, 0, 0)
        container.BackgroundTransparency = 1
        container.Visible = false
        container.ZIndex = 2

        local containerLayout = Instance.new("UIListLayout")
        containerLayout.Parent = container
        containerLayout.SortOrder = Enum.SortOrder.LayoutOrder
        containerLayout.Padding = UDim(0, 12)

        -- Kullanıcının içerik oluşturma callback'i
        local function buildContent()
            -- Önceki içeriği temizle
            for _, child in ipairs(container:GetChildren()) do
                if child:IsA("UIListLayout") == false then
                    child:Destroy()
                end
            end
            -- Callback'i çağır, içine group oluşturma fonksiyonlarını ver
            if contentCallback then
                contentCallback(container, containerLayout)
            end
        end

        -- Butona tıklanınca tab'ı göster
        btn.MouseButton1Click:Connect(function()
            if currentTab == btn then return end

            -- Önceki tab'i gizle
            if currentTab then
                local oldBtn = currentTab
                local oldContainer = tabContents[oldBtn]
                if oldContainer then oldContainer.Visible = false end
                tween(oldBtn, TweenInfo.new(0.2), {BackgroundTransparency = 0.8}):Play()
                oldBtn.TextColor3 = colors.textDim
                tween(oldBtn.Indicator, TweenInfo.new(0.2), {BackgroundTransparency = 1}):Play()
            end

            -- Yeni tab'i göster
            currentTab = btn
            local cont = tabContents[btn]
            if not cont then
                -- İlk kez açılıyorsa içeriği oluştur
                buildContent()
                cont = container
                tabContents[btn] = cont
            end
            cont.Visible = true
            tween(btn, TweenInfo.new(0.2), {BackgroundTransparency = 0.3}):Play()
            btn.TextColor3 = colors.accent
            tween(btn.Indicator, TweenInfo.new(0.2), {BackgroundTransparency = 0}):Play()

            -- Canvas boyutunu güncelle
            task.wait(0.05)
            contentArea.CanvasSize = UDim2(0, 0, 0, contentLayout.AbsoluteContentSize.Y + 20)
        end)

        -- İlk tab'ı otomatik seç
        if #tabButtons == 0 then
            task.spawn(function()
                btn.MouseButton1Click:Fire()
            end)
        end

        table.insert(tabButtons, btn)
        return btn
    end

            hit.Parent = switchBg
            hit.Size = UDim2(1, 0, 1, 0)
            hit.BackgroundTransparency = 1
            hit.Text = ""

            hit.MouseButton1Click:Connect(function()
                obj.Value = not obj.Value
                if obj.Value then
                    tween(thumb, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                        Position = UDim2(1, -27, 0.5, -12),
                        BackgroundColor3 = colors.toggleOn
                    }):Play()
                    status.Text = "ON"
                    status.TextColor3 = colors.toggleOn
                else
                    tween(thumb, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                        Position = UDim2(0, 3, 0.5, -12),
                        BackgroundColor3 = colors.toggleOff
                    }):Play()
                    status.Text = "OFF"
                    status.TextColor3 = colors.toggleOff
                end
                if callback then pcall(callback, obj.Value) end
            end)

            obj.Set = function(self, val)
                self.Value = val
                if val then
                    thumb.Position = UDim2(1, -27, 0.5, -12)
                    thumb.BackgroundColor3 = colors.toggleOn
                    status.Text = "ON"
                    status.TextColor3 = colors.toggleOn
                else
                    thumb.Position = UDim2(0, 3, 0.5, -12)
                    thumb.BackgroundColor3 = colors.toggleOff
                    status.Text = "OFF"
                    status.TextColor3 = colors.toggleOff
                end
                if callback then pcall(callback, val) end
            end
            return obj
        end

        -- Slider
        function group:CreateSlider(parent, label, min, max, default, callback)
            min = min or 0
            max = max or 100
            default = default or 50
            local obj = { Value = default }

            local frame = Instance.new("Frame")
            frame.Parent = parent
            frame.Size = UDim2(1, 0, 0, 60)
            frame.BackgroundColor3 = colors.elemBg
            corner(frame, 8)
            stroke(frame, colors.elemBorder, 1, 0.4)

            local lbl = Instance.new("TextLabel")
            lbl.Parent = frame
            lbl.Size = UDim2(1, 0, 0, 26)
            lbl.Position = UDim2(0, 12, 0, 6)
            lbl.BackgroundTransparency = 1
            lbl.Text = label .. " (" .. tostring(default) .. ")"
            lbl.Font = Enum.Font.GothamMedium
            lbl.TextSize = 16
            lbl.TextColor3 = colors.text
            lbl.TextXAlignment = Enum.TextXAlignment.Left

            local bg = Instance.new("Frame")
            bg.Parent = frame
            bg.Size = UDim2(1, -24, 0, 8)
            bg.Position = UDim2(0, 12, 0, 38)
            bg.BackgroundColor3 = colors.sliderBg
            corner(bg, 4)

            local fill = Instance.new("Frame")
            fill.Parent = bg
            fill.Size = UDim2((default - min) / (max - min), 0, 1, 0)
            fill.BackgroundColor3 = colors.sliderFill
            corner(fill, 4)

            local thumb = Instance.new("Frame")
            thumb.Parent = fill
            thumb.Size = UDim2(0, 18, 0, 18)
            thumb.Position = UDim2(1, -9, 0.5, -9)
            thumb.BackgroundColor3 = colors.accent
            corner(thumb, 9)
            stroke(thumb, colors.shadow, 2, 0.3)

            local hit = Instance.new("TextButton")
            hit.Parent = bg
            hit.Size = UDim2(1, 0, 1, 8)
            hit.BackgroundTransparency = 1
            hit.Text = ""

            local dragging = false
            local function update(posX)
                local rel = math.clamp((posX - bg.AbsolutePosition.X) / bg.AbsoluteSize.X, 0, 1)
                fill.Size = UDim2(rel, 0, 1, 0)
                local val = math.floor(min + (max - min) * rel)
                obj.Value = val
                lbl.Text = label .. " (" .. tostring(val) .. ")"
                if callback then pcall(callback, val) end
            end

            hit.InputBegan:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1 then
                    dragging = true
                    update(input.Position.X)
                end
            end)
            UserInput.InputEnded:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1 then
                    dragging = false
                end
            end)
            UserInput.InputChanged:Connect(function(input)
                if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
                    update(input.Position.X)
                end
            end)

            obj.Set = function(self, val)
                val = math.clamp(val, min, max)
                self.Value = val
                local rel = (val - min) / (max - min)
                fill.Size = UDim2(rel, 0, 1, 0)
                lbl.Text = label .. " (" .. tostring(val) .. ")"
                if callback then pcall(callback, val) end
            end
            return obj
        end

            list.Position = UDim2(0, 5, 0, 44)
            list.BackgroundTransparency = 1
            list.BorderSizePixel = 0
            list.ScrollBarThickness = 4
            list.ScrollBarImageColor3 = colors.accent
            list.CanvasSize = UDim2(0, 0, 0, 0)

            local listLayout = Instance.new("UIListLayout")
            listLayout.Parent = list
            listLayout.SortOrder = Enum.SortOrder.LayoutOrder
            listLayout.Padding = UDim(0, 4)

            listLayout:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(function()
                list.CanvasSize = UDim2(0, 0, 0, listLayout.AbsoluteContentSize.Y)
            end)

            local toggleBtn = Instance.new("TextButton")
            toggleBtn.Parent = frame
            toggleBtn.Size = UDim2(1, 0, 1, 0)
            toggleBtn.BackgroundTransparency = 1
            toggleBtn.Text = ""

            toggleBtn.MouseButton1Click:Connect(function()
                expanded = not expanded
                if expanded then
                    tween(frame, TweenInfo.new(0.25), {Size = UDim2(1, 0, 0, 44 + math.min(#items * 30, 120))}):Play()
                    arrow.Text = "▲"
                    list.Size = UDim2(1, -10, 0, math.min(#items * 30, 120) - 10)
                else
                    tween(frame, TweenInfo.new(0.25), {Size = UDim2(1, 0, 0, 44)}):Play()
                    arrow.Text = "▼"
                    list.Size = UDim2(1, -10, 0, 0)
                end
            end)

            for _, item in ipairs(items) do
                local itemBtn = Instance.new("TextButton")
                itemBtn.Parent = list
                itemBtn.Size = UDim2(1, 0, 0, 30)
                itemBtn.BackgroundColor3 = colors.bgSide
                itemBtn.BackgroundTransparency = 0.5
                itemBtn.Text = item
                itemBtn.Font = Enum.Font.GothamMedium
                itemBtn.TextSize = 14
                itemBtn.TextColor3 = colors.text
                corner(itemBtn, 4)
                stroke(itemBtn, colors.elemBorder, 1, 0.3)

                itemBtn.MouseButton1Click:Connect(function()
                    obj.Selected = item
                    valueLbl.Text = item
                    expanded = false
                    tween(frame, TweenInfo.new(0.25), {Size = UDim2(1, 0, 0, 44)}):Play()
                    arrow.Text = "▼"
                    list.Size = UDim2(1, -10, 0, 0)
                    if callback then pcall(callback, item) end
                end)
            end

            obj.Select = function(self, item)
                for _, v in ipairs(items) do
                    if v == item then
                        self.Selected = item
                        valueLbl.Text = item
                        if callback then pcall(callback, item) end
                        break
                    end
                end
            end
            return obj
        end

        function group:CreateLabel(parent, text)
            local lbl = Instance.new("TextLabel")

            lbl.Size = UDim2(1, 0, 0, 30)
            lbl.BackgroundTransparency = 1

            lbl.Font = Enum.Font.GothamMedium

            stroke(frame, colors.elemBorder, 1, 0.4)

                    math.random(50, 255),
                    math.random(50, 255)
                )
                box.BackgroundColor3 = newColor
                if callback then pcall(callback, newColor) end

        -- Keybind
        function group:CreateKeybind(parent, label, default, callback)
            local frame = Instance.new("Frame")
            frame.Parent = parent
            frame.Size = UDim2(1, 0, 0, 44)
            frame.BackgroundColor3 = colors.elemBg
            corner(frame, 8)
            stroke(frame, colors.elemBorder, 1, 0.4)

            local lbl = Instance.new("TextLabel")
            lbl.Parent = frame
            lbl.Size = UDim2(0.6, 0, 1, 0)
            lbl.Position = UDim2(0, 12, 0, 0)
            lbl.BackgroundTransparency = 1
            lbl.Text = label
            lbl.Font = Enum.Font.GothamMedium
            lbl.TextSize = 16
            lbl.TextColor3 = colors.text
            lbl.TextXAlignment = Enum.TextXAlignment.Left

            local keyBtn = Instance.new("TextButton")
            keyBtn.Parent = frame
            keyBtn.Size = UDim2(0, 80, 0, 34)
            keyBtn.Position = UDim2(1, -92, 0.5, -17)
            keyBtn.BackgroundColor3 = colors.bg
            keyBtn.Text = default or "None"
            keyBtn.Font = Enum.Font.GothamBold
            keyBtn.TextSize = 14
            keyBtn.TextColor3 = colors.accent
            corner(keyBtn, 6)
            stroke(keyBtn, colors.elemBorder, 1, 0.5)

            local listening = false
            keyBtn.MouseButton1Click:Connect(function()
                if listening then return end
                listening = true
                keyBtn.Text = "..."
                keyBtn.TextColor3 = colors.textDim

                local conn
                conn = UserInput.InputBegan:Connect(function(input, processed)
                    if processed then return end
                    local key = input.KeyCode.Name
                    if key ~= "Unknown" then
                        listening = false
                        keyBtn.Text = key
                        keyBtn.TextColor3 = colors.accent
                        if callback then pcall(callback, key) end
                        conn:Disconnect()
                    end
                end)
            end)
            return keyBtn
        end

        return group
    end

    screenGui.Enabled = true
